# PassThrough Realism - 30 second gameplay film (format: see Rendering/Gameplay.cs).
# Record (test client running on a copy of the world, see README):
#   dotnet run --project Realism -c Release -- --live --no-launch --gameplay Realism/gameplay.txt --out gameplay.mp4
# Look angles are Minecraft's: yaw 0 = south, 90 = west, 180 = north, -90 = east; pitch positive = down.

# 1. A sunny walk along the open strip of meadow, between the grazing animals.
seg 8 view=third
cmd difficulty peaceful
cmd gamemode survival
cmd gamerule doDaylightCycle false
cmd time set 4500
cmd weather clear
cmd effect give @s fire_resistance infinite 0 true
cmd effect give @s resistance infinite 4 true
cmd tp @s 418.5 92 -76.5 -90 4
cmd summon cow 426.5 75 -74.5 {NoAI:1b,Rotation:[60f,0f]}
cmd summon sheep 428.5 77 -78.5 {NoAI:1b,Rotation:[100f,0f]}
cmd summon pig 425.5 77 -79.5 {NoAI:1b,Rotation:[150f,0f]}
cmd summon chicken 422.5 75 -74.6 {NoAI:1b,Rotation:[20f,0f]}
cmd summon chicken 420.6 75 -74.4 {NoAI:1b,Rotation:[-40f,0f]}
cmd summon wolf 427.5 77 -80.2 {NoAI:1b,Rotation:[160f,0f]}
at 0 look -90 4
at 0 cam 0 4.4 0.55
at 0.6 keys forward
at 2.1 keys none
at 3 look -90 6
at 8 look -105 8
at 2.5 cam 0 4.4 0.55
at 8 cam -50 4.2 0.6
